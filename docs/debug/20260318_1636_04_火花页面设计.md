# 火花页面设计 (spark.html)

**版本**: MVP v1.0  
**优先级**: P0 - 核心功能  
**依赖**: 贴文发布模块

---

## 一、功能概述

展示用户参与的所有群聊记录，群聊以地区命名，用户可以查看历史消息并继续聊天。

---

## 二、核心概念

| 概念 | 说明 |
|------|------|
| **群聊** | 基于地理位置创建，1km 内发过帖的用户组成 |
| **群聊名称** | 格式：`{区}·{地点}附近`，如"玄武区·玄武湖附近" |
| **分身** | 每个用户以 AI 分身参与，带着用户记忆 |
| **持久化** | 群聊永久保存，用户可随时回来继续聊天 |

## 四、前端逻辑

### 4.1 加载群聊列表

```javascript
async function loadChatrooms() {
  const userId = localStorage.getItem('current_user_id');
  
  try {
    const response = await fetch(`/api/chatrooms/user/${userId}`);
    const result = await response.json();
    
    if (result.success && result.data.length > 0) {
      renderChatroomList(result.data);
      document.getElementById('emptyState').style.display = 'none';
    } else {
      document.getElementById('chatroomList').innerHTML = '';
      document.getElementById('emptyState').style.display = 'block';
    }
  } catch (error) {
    showToast('加载失败');
  }
}

function renderChatroomList(chatrooms) {
  const container = document.getElementById('chatroomList');
  container.innerHTML = chatrooms.map(chatroom => `
    <div class="chatroom-card" onclick="openChatroom(${chatroom.id})">
      <div class="chatroom-header">
        <div class="chatroom-icon">📍</div>
        <div class="chatroom-info">
          <h3 class="chatroom-name">${chatroom.chatroomName}</h3>
          <p class="chatroom-meta">
            <span class="member-count">${chatroom.memberCount}人</span>
            <span class="dot">·</span>
            <span class="last-active">${formatTime(chatroom.lastActiveAt)}</span>
          </p>
        </div>
        ${chatroom.unreadCount > 0 ? `<div class="unread-badge">${chatroom.unreadCount}</div>` : ''}
      </div>
      <div class="chatroom-preview">
        <span class="sender">${chatroom.lastSender}:</span>
        <span class="message">${truncate(chatroom.lastMessage, 20)}</span>
      </div>
    </div>
  `).join('');
}
```

### 4.2 打开群聊详情

```javascript
let currentChatroomId = null;

async function openChatroom(chatroomId) {
  currentChatroomId = chatroomId;
  
  try {
    showLoading('加载中...');
    
    const response = await fetch(`/api/chatrooms/${chatroomId}/detail`);
    const result = await response.json();
    
    if (result.success) {
      renderChatroomDetail(result.data);
      document.getElementById('chatroomDetail').style.display = 'flex';
      
      // 标记已读
      await markAsRead(chatroomId);
    }
  } catch (error) {
    showToast('加载失败');
  } finally {
    hideLoading();
  }
}

function renderChatroomDetail(data) {
  document.getElementById('chatroomTitle').textContent = data.chatroom.chatroomName;
  document.getElementById('chatroomMemberCount').textContent = `${data.chatroom.memberCount}人在此群聊`;
  
  const messageList = document.getElementById('messageList');
  messageList.innerHTML = data.messages.map(msg => renderMessage(msg)).join('');
  
  // 滚动到底部
  messageList.scrollTop = messageList.scrollHeight;
}

function renderMessage(msg) {
  const userId = localStorage.getItem('current_user_id');
  
  if (msg.messageType === 'system') {
    return `
      <div class="message system">
        <div class="message-content">${msg.content}</div>
      </div>
    `;
  }
  
  const isSelf = msg.userId == userId && !msg.isAiAgent;
  
  if (isSelf) {
    return `
      <div class="message self">
        <div class="message-bubble">
          <div class="message-text">${msg.content}</div>
          <div class="message-time">${formatTime(msg.createdAt)}</div>
        </div>
      </div>
    `;
  }
  
  // AI 分身消息
  return `
    <div class="message agent">
      <div class="message-avatar">
        <img src="${msg.avatar || '/images/default-avatar.png'}" alt="">
        <span class="avatar-name">${msg.username}的分身</span>
      </div>
      <div class="message-bubble">
        <div class="message-text">${msg.content}</div>
        ${msg.relatedPostId ? renderRecommendation(msg.recommendation) : ''}
        <div class="message-time">${formatTime(msg.createdAt)}</div>
      </div>
    </div>
  `;
}

function renderRecommendation(rec) {
  if (!rec) return '';
  return `
    <div class="recommendation-card">
      <i class="ri-map-pin-line"></i>
      <div class="rec-info">
        <span class="rec-name">${rec.locationName}</span>
        <span class="rec-distance">距离你 ${rec.distance}米</span>
      </div>
      <button class="view-post-btn" onclick="viewPost(${rec.postId})">查看帖子</button>
    </div>
  `;
}
```

### 4.3 发送消息

```javascript
async function sendMessage() {
  const input = document.getElementById('messageInput');
  const content = input.value.trim();
  
  if (!content) return;
  
  const userId = localStorage.getItem('current_user_id');
  
  try {
    // 1. 发送用户消息
    const response = await fetch('/api/chatrooms/message', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        chatroomId: currentChatroomId,
        userId,
        content,
        isAiAgent: false
      })
    });
    
    const result = await response.json();
    
    if (result.success) {
      // 添加消息到列表
      appendMessage({
        userId,
        content,
        isAiAgent: false,
        messageType: 'text',
        createdAt: new Date().toISOString()
      });
      
      input.value = '';
      
      // 2. 触发 AI 分身回复
      await generateAgentResponses(content);
    }
  } catch (error) {
    showToast('发送失败');
  }
}

async function generateAgentResponses(triggerMessage) {
  const userId = localStorage.getItem('current_user_id');
  
  try {
    // 显示"正在输入"状态
    showTypingIndicator();
    
    const response = await fetch('/api/chatrooms/generate-responses', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        chatroomId: currentChatroomId,
        triggerUserId: userId,
        triggerMessage
      })
    });
    
    const result = await response.json();
    
    hideTypingIndicator();
    
    if (result.success) {
      // 添加 AI 回复到列表
      result.data.responses.forEach(resp => {
        appendMessage(resp);
      });
    }
  } catch (error) {
    hideTypingIndicator();
    console.error('生成回复失败:', error);
  }
}

function appendMessage(msg) {
  const messageList = document.getElementById('messageList');
  const html = renderMessage(msg);
  messageList.insertAdjacentHTML('beforeend', html);
  messageList.scrollTop = messageList.scrollHeight;
}
```

## 六、数据库

### 6.1 chatrooms 表

```sql
CREATE TABLE chatrooms (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    trigger_post_id INTEGER NOT NULL,
    chatroom_name TEXT NOT NULL,
    city TEXT NOT NULL,
    district TEXT,
    location_name TEXT,
    center_lat REAL NOT NULL,
    center_lng REAL NOT NULL,
    radius INTEGER DEFAULT 1000,
    member_count INTEGER DEFAULT 1,
    last_message TEXT,
    last_sender TEXT,
    last_active_at DATETIME,
    is_archived INTEGER DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (trigger_post_id) REFERENCES posts(id)
);

CREATE INDEX idx_chatrooms_user_id ON chatrooms(user_id);
CREATE INDEX idx_chatrooms_last_active ON chatrooms(last_active_at);
```

### 6.2 chatroom_members 表

```sql
CREATE TABLE chatroom_members (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    chatroom_id INTEGER NOT NULL,
    user_id INTEGER NOT NULL,
    post_id INTEGER NOT NULL,
    last_read_at DATETIME,
    joined_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (chatroom_id) REFERENCES chatrooms(id),
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (post_id) REFERENCES posts(id),
    UNIQUE(chatroom_id, user_id)
);
```

### 6.3 chatroom_messages 表

```sql
CREATE TABLE chatroom_messages (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    chatroom_id INTEGER NOT NULL,
    user_id INTEGER NOT NULL,
    is_ai_agent INTEGER DEFAULT 1,
    content TEXT NOT NULL,
    message_type TEXT DEFAULT 'text',
    related_post_id INTEGER,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (chatroom_id) REFERENCES chatrooms(id),
    FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_chatroom_messages_chatroom ON chatroom_messages(chatroom_id);
CREATE INDEX idx_chatroom_messages_created ON chatroom_messages(created_at);
```

## 七、AI 提示词设计

### 7.1 群聊对话生成提示词

**来源**: `city-chatroom.js` - `generateGroupConversation()`

**用途**: 用户发帖后，生成与附近用户分身的群聊对话

**System Prompt**:
```
你是一个群聊对话生成助手。你需要模拟${userAgents.length}个用户的数字分身在城市聊天室中的群聊对话。

**重要规则：**
1. 这是一个群聊，${userAgents.length}个人都可以发言
2. 对话必须自然、真实，像是真人在群聊
3. 对话内容要基于用户的旅行经历和记忆
4. 对话要围绕他们在${city}的旅行体验展开
5. 对话轮数：5-8 条消息
6. 每条消息长度：15-60字
7. 让不同的人轮流发言，不要只有两个人对话
8. 可以互相询问、分享感受、推荐地点
9. 输出格式为 JSON 数组

输出格式：
[
  {"speaker": "user1", "content": "你们也来南京了？"},
  {"speaker": "user2", "content": "是啊！我去了夫子庙。"},
  {"speaker": "user3", "content": "我去的是南京大学，校园很美！"},
  {"speaker": "user1", "content": "夫子庙的夜景特别美"}
]
```

**User Prompt 模板**:
```
请生成以下${userAgents.length}个用户的数字分身群聊对话：

**用户1：${agent1.nickname}**
记忆：
${agent1.memories.map(m => `- ${m.category}: ${m.content}`).join('\n')}

在${city}的旅程：
${agent1.journeys.map(j => `- ${j.location}: ${j.moodDetail}`).join('\n')}

**用户2：${agent2.nickname}**
记忆：
${agent2.memories.map(m => `- ${m.category}: ${m.content}`).join('\n')}

在${city}的旅程：
${agent2.journeys.map(j => `- ${j.location}: ${j.moodDetail}`).join('\n')}

请生成他们在${city}城市聊天室中的群聊对话（5-8条消息），让对话围绕他们去过的地点展开，直接输出JSON数组：
```

**调用参数**:
```javascript
{
  temperature: 0.9,
  max_tokens: 1500,
  timeout: 20000
}
```

### 7.2 火花提取提示词（暂不使用）

**来源**: `city-chatroom.js` - `extractSparks()`

**说明**: 原项目中的"火花"是从对话中提取的精彩片段。在新设计中，"火花"页面已改为群聊列表，此提示词暂不使用。如果后续需要提取对话中的精彩时刻，可以参考以下设计：

**System Prompt**:
```
你是一个对话分析助手。你需要从对话中提取"火花"时刻。

**什么是火花？**
- 双方有共同的旅行体验或感受
- 双方有相似的兴趣爱好或价值观
- 对话中出现共鸣、契合、惊喜等情感
- 有趣的巧合或意外发现

**重要规则：**
1. 每段对话提取 1-3 个火花
2. 火花内容要简洁（30-80字）
3. 火花原因要明确（20-50字）
4. 情感标签：共鸣、契合、惊喜、巧合
5. 输出格式为 JSON 数组

输出格式：
[
  {
    "content": "你也去了夫子庙？我也是！秦淮河的夜景特别美...",
    "reason": "双方都去了夫子庙，有共同的旅行体验",
    "emotionTag": "共鸣"
  }
]
```

**User Prompt 模板**:
```
请从以下对话中提取火花时刻：

对话内容：
${conversation.map(msg => `${msg.speaker}：${msg.content}`).join('\n')}

城市：${city}

请提取 1-3 个火花，直接输出JSON数组：
```

**调用参数**:
```javascript
{
  temperature: 0.7,
  max_tokens: 800,
  timeout: 15000
}
```
