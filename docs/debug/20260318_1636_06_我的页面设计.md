## 四、前端逻辑

### 4.1 加载用户信息

```javascript
async function loadUserInfo() {
  const userInfo = JSON.parse(localStorage.getItem('user_info') || '{}');
  
  if (userInfo.id) {
    document.getElementById('userNickname').textContent = userInfo.nickname || userInfo.username;
    document.getElementById('userBio').textContent = userInfo.bio || '这个人很懒，什么都没写';
    document.getElementById('userAvatar').src = userInfo.avatar || '/images/default-avatar.png';
  }
}
```

### 4.2 保存个人信息

```javascript
async function saveProfile() {
  const nickname = document.getElementById('editNickname').value.trim();
  const bio = document.getElementById('editBio').value.trim();
  const userId = localStorage.getItem('current_user_id');
  
  try {
    showLoading('保存中...');
    
    const response = await fetch(`/api/users/${userId}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ nickname, bio })
    });
    
    const result = await response.json();
    
    if (result.success) {
      // 更新本地存储
      const userInfo = JSON.parse(localStorage.getItem('user_info') || '{}');
      userInfo.nickname = nickname;
      userInfo.bio = bio;
      localStorage.setItem('user_info', JSON.stringify(userInfo));
      
      // 更新页面显示
      document.getElementById('userNickname').textContent = nickname;
      document.getElementById('userBio').textContent = bio || '这个人很懒，什么都没写';
      
      hideEditModal();
      showToast('保存成功');
    }
  } catch (error) {
    showToast('保存失败');
  } finally {
    hideLoading();
  }
}
```

### 4.3 退出登录

```javascript
function logout() {
  if (confirm('确定要退出登录吗？')) {
    localStorage.removeItem('current_user_id');
    localStorage.removeItem('user_info');
    localStorage.removeItem('current_plan_id');
    localStorage.removeItem('travel_status');
    
    window.location.href = '/pages/mobile/login.html';
  }
}
```
