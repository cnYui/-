**版本**: MVP v1.0

## 一、功能概述

应用的核心页面，展示高德地图，提供"开始旅行"入口，显示旅行进度和贴文标记。

## 三、用户流程

### 3.1 开始旅行流程

```
用户点击"开始旅行"
    ↓
弹出旅行表单
    ↓
输入：出发地、目的地、旅行方式
    ↓
点击"生成计划"
    ↓
后端调用 DeepSeek 生成计划
    ↓
返回按天数划分的景点列表
    ↓
前端展示旅行计划卡片
    ↓
用户点击"开始"
    ↓
进入旅行进度推进模式
```

### 3.2 旅行进度流程

```
旅行开始
    ↓
显示当前景点 + 倒计时
    ↓
倒计时结束 → 切换下一景点
    ↓
每天结束 → 生成日记（发送到收信箱）
    ↓
全部结束 → 显示旅行完成
```

### 3.3 贴文发布流程

```
用户点击"+"按钮
    ↓
选择：文字 / 图片
    ↓
├── 文字 → 输入内容 → 发布
└── 图片 → 拍照/选择 → 添加描述 → 发布
    ↓
贴文保存到数据库
    ↓
地图上显示表情标记
    ↓
触发城市聊天室匹配（跳转到 Spark 页面）
```

let map;

### 5.2 生成旅行计划

```javascript
async function generatePlan() {
  const departure = document.getElementById('departureInput').value;
  const destination = document.getElementById('destinationInput').value;
  const travelMode = document.getElementById('travelModeSelect').value;
  const userId = localStorage.getItem('current_user_id');
  
  if (!departure || !destination) {
    showToast('请填写出发地和目的地');
    return;
  }
  
  try {
    showLoading('AI 正在生成旅行计划...');
    
    const response = await fetch('/api/travel/plan/generate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ userId, departure, destination, travelMode })
    });
    
    const result = await response.json();
    
    if (result.success) {
      displayPlan(result.data);
    } else {
      showToast(result.error || '生成失败');
    }
  } catch (error) {
    showToast('网络错误');
  } finally {
    hideLoading();
  }
}
```

### 5.3 展示旅行计划

```javascript
function displayPlan(planData) {
  const container = document.getElementById('planCards');
  container.innerHTML = '';
  
  // 按天分组
  const dayGroups = {};
  planData.dailyPlans.forEach(spot => {
    if (!dayGroups[spot.day]) dayGroups[spot.day] = [];
    dayGroups[spot.day].push(spot);
  });
  
  // 生成卡片
  Object.keys(dayGroups).forEach(day => {
    const dayCard = document.createElement('div');
    dayCard.className = 'day-card';
    dayCard.innerHTML = `
      <h3>第 ${day} 天</h3>
      <ul>
        ${dayGroups[day].map(spot => `
          <li>
            <span class="spot-name">${spot.location}</span>
            <span class="spot-duration">${spot.duration}分钟</span>
          </li>
        `).join('')}
      </ul>
    `;
    container.appendChild(dayCard);
  });
  
  // 存储计划数据
  window.currentPlan = planData;
  
  // 显示计划
  document.getElementById('travelFormModal').style.display = 'none';
  document.getElementById('travelPlanDisplay').style.display = 'block';
}
```

### 5.4 开始旅行

```javascript
async function startTravel() {
  const userId = localStorage.getItem('current_user_id');
  const plan = window.currentPlan;
  
  try {
    showLoading('开始旅行...');
    
    const response = await fetch('/api/travel/plan/start', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        userId,
        destination: plan.destination,
        departure: plan.departure,
        travelMode: plan.travelMode,
        estimatedDays: plan.estimatedDays,
        dailyPlans: plan.dailyPlans
      })
    });
    
    const result = await response.json();
    
    if (result.success) {
      // 存储旅行状态
      localStorage.setItem('current_plan_id', result.data.planId);
      localStorage.setItem('travel_status', 'traveling');
      
      // 切换到旅行模式
      enterTravelMode(result.data);
    }
  } catch (error) {
    showToast('开始旅行失败');
  } finally {
    hideLoading();
  }
}
```

### 5.5 旅行进度推进

```javascript
let countdownTimer;

function enterTravelMode(progressData) {
  document.getElementById('travelPlanDisplay').style.display = 'none';
  document.querySelector('.action-idle').style.display = 'none';
  document.querySelector('.action-traveling').style.display = 'block';
  
  updateCurrentSpot(progressData);
  startCountdown(progressData.remainingSeconds);
}

function updateCurrentSpot(progress) {
  const spot = progress.currentSpot;
  document.getElementById('spotName').textContent = spot.location;
  document.getElementById('spotDesc').textContent = spot.description;
  
  // 地图定位到当前景点
  if (spot.lat && spot.lng) {
    map.setCenter([spot.lng, spot.lat]);
    addMarker(spot);
  }
}

function startCountdown(seconds) {
  let remaining = seconds;
  
  countdownTimer = setInterval(() => {
    remaining--;
    
    const mins = Math.floor(remaining / 60);
    const secs = remaining % 60;
    document.getElementById('countdown').textContent = 
      `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
    
    if (remaining <= 0) {
      clearInterval(countdownTimer);
      onSpotComplete();
    }
  }, 1000);
}

async function onSpotComplete() {
  // 获取最新进度
  const userId = localStorage.getItem('current_user_id');
  const response = await fetch(`/api/travel/progress/${userId}`);
  const result = await response.json();
  
  if (result.data.status === 'completed') {
    onTravelComplete();
  } else {
    updateCurrentSpot(result.data);
    startCountdown(result.data.remainingSeconds);
  }
}
```

## 七、AI 提示词设计

### 7.1 旅行计划生成提示词

**来源**: `travel-plan-generator.js`

**System Prompt**:
```
你是一个旅行规划助手。你必须只输出纯JSON格式的旅行计划，不要输出任何思考过程、解释或分析。直接输出JSON对象，不要有任何其他文字。
```

**User Prompt 模板**:
```
你是一个专业的旅行规划助手。请根据以下信息为用户制定完整的旅行计划，包括往返路程。

【用户信息】
${memoriesText || '暂无'}

【旅行信息】
- 出发地：${departure}
- 目的地：${destination}
- 出行方式：${travelMode}

【重要规则】
1. 必须考虑往返路程时间：
   - 第1天或前半天：从${departure}前往${destination}的路程（根据距离和${travelMode}估算时间）
   - 最后1天或后半天：从${destination}返回${departure}的路程
   - 如果距离很远（跨省/跨国），往返各占半天到一天
   - 如果距离较近（同城/邻近），往返各占2-4小时

2. 游玩时间安排：
   - 每天游玩时间：最多10小时（600分钟）
   - 每天景点数量：3-5个
   - 单个景点停留时间：60-180分钟

3. 预估天数：1-7天（包含往返路程）
4. 行程要符合用户的兴趣偏好

【输出格式】
严格按照以下 JSON 格式输出：

{"estimatedDays":天数,"dailyPlans":[{"day":天数,"location":"地点名称","description":"活动描述","duration":分钟数}]}

【示例】
北京到杭州（飞机，3天）：
{"estimatedDays":3,"dailyPlans":[
  {"day":1,"location":"首都国际机场-杭州萧山机场","description":"乘坐飞机前往杭州","duration":120},
  {"day":1,"location":"西湖","description":"游览西湖风光","duration":180},
  {"day":2,"location":"灵隐寺","description":"参观古刹","duration":120},
  {"day":2,"location":"宋城","description":"观看演出","duration":180},
  {"day":3,"location":"河坊街","description":"品尝美食","duration":120},
  {"day":3,"location":"杭州萧山机场-首都国际机场","description":"返程回北京","duration":120}
]}

请根据出发地"${departure}"、目的地"${destination}"、出行方式"${travelMode}"和用户记忆生成完整旅行计划，直接输出JSON：
```
