# 收信箱设计 (notes.html)

**版本**: MVP v1.0  
**优先级**: P1 - 重要功能  
**依赖**: 旅行模块、日记生成

---

## 一、功能概述

展示用户收到的所有邮件，包括每日旅行日记、AI分身发现的推荐、好友消息等。

---

## 二、邮件类型

| 类型 | 说明 | MVP |
|------|------|-----|
| `diary` | 每日旅行日记 | ✅ |
| `discovery` | AI分身发现的推荐（玩法1） | ✅ |
| `selfie` | 自拍照片 | ❌ v2.0 |
| `gift` | 伴手礼 | ❌ v2.0 |
| `friend_greeting` | 好友问候 | ❌ v2.0 |
| `friend_photo` | 好友合影 | ❌ v2.0 |

## 三、用户流程

### 3.1 查看邮件列表

```
用户进入收信箱
    ↓
加载邮件列表（按时间倒序）
    ↓
显示未读/已读状态
    ↓
点击邮件 → 查看详情
    ↓
标记为已读
```

### 3.2 AI分身发现推荐（玩法1）

```
用户开始旅行 / 空投分身到城市
    ↓
后端启动 AI 分身探索任务
    ↓
AI 扫描城市历史帖子
    ↓
根据用户记忆匹配高度相关的地点
    ↓
生成推荐邮件发送到收信箱
    ↓
用户查看邮件
    ↓
点击"加入行程" → 更新旅行计划
```

## 五、前端逻辑

### 5.1 加载邮件列表

```javascript
async function loadMails(filter = 'all') {
  const userId = localStorage.getItem('current_user_id');
  
  try {
    let url = `/api/mails/user/${userId}`;
    if (filter !== 'all') {
      url += `?type=${filter}`;
    }
    
    const response = await fetch(url);
    const result = await response.json();
    
    if (result.success && result.data.length > 0) {
      renderMailList(result.data);
      document.getElementById('emptyState').style.display = 'none';
    } else {
      document.getElementById('mailList').innerHTML = '';
      document.getElementById('emptyState').style.display = 'block';
    }
  } catch (error) {
    showToast('加载失败');
  }
}

function renderMailList(mails) {
  const container = document.getElementById('mailList');
  container.innerHTML = mails.map(mail => {
    const icon = mail.mailType === 'diary' ? '📔' : '✨';
    const typeClass = mail.mailType;
    
    return `
      <div class="mail-card ${typeClass}" onclick="openMail(${mail.id})">
        <div class="mail-icon">${icon}</div>
        <div class="mail-content">
          <div class="mail-header">
            <h3 class="mail-title">${mail.title}</h3>
            <span class="mail-time">${formatTime(mail.createdAt)}</span>
          </div>
          <p class="mail-preview">${truncate(mail.content, 30)}</p>
        </div>
        ${mail.isRead ? '' : '<div class="unread-dot"></div>'}
      </div>
    `;
  }).join('');
}
```

### 5.2 查看邮件详情

```javascript
async function openMail(mailId) {
  try {
    showLoading('加载中...');
    
    const response = await fetch(`/api/mails/${mailId}`);
    const result = await response.json();
    
    if (result.success) {
      renderMailDetail(result.data);
      document.getElementById('mailDetail').style.display = 'flex';
      
      // 标记已读
      if (!result.data.isRead) {
        await markMailAsRead(mailId);
      }
    }
  } catch (error) {
    showToast('加载失败');
  } finally {
    hideLoading();
  }
}

function renderMailDetail(mail) {
  document.getElementById('mailTitle').textContent = mail.title;
  
  const content = document.getElementById('mailContent');
  
  if (mail.mailType === 'diary') {
    content.innerHTML = renderDiaryContent(mail);
  } else if (mail.mailType === 'discovery') {
    content.innerHTML = renderDiscoveryContent(mail);
  }
}

function renderDiaryContent(mail) {
  const data = JSON.parse(mail.extraData || '{}');
  
  return `
    <div class="diary-content">
      <div class="diary-date">${formatDate(mail.createdAt)}</div>
      <div class="diary-text">${mail.content}</div>
      ${mail.imageUrl ? `<div class="diary-image"><img src="${mail.imageUrl}" alt=""></div>` : ''}
      ${data.visitedPlaces ? `
        <div class="diary-places">
          <h4>今日足迹</h4>
          <ul>
            ${data.visitedPlaces.map(p => `<li>${p}</li>`).join('')}
          </ul>
        </div>
      ` : ''}
    </div>
  `;
}

function renderDiscoveryContent(mail) {
  const data = JSON.parse(mail.extraData || '{}');
  
  return `
    <div class="discovery-content">
      <div class="discovery-message">${mail.content}</div>
      ${data.relatedPost ? `
        <div class="discovery-post">
          <div class="post-header">
            <img class="post-avatar" src="${data.relatedPost.avatar || '/images/default-avatar.png'}" alt="">
            <div class="post-info">
              <span class="post-author">${data.relatedPost.username}</span>
              <span class="post-time">${data.relatedPost.timeAgo}</span>
            </div>
          </div>
          <div class="post-content">${data.relatedPost.content}</div>
          ${data.relatedPost.imageUrl ? `<div class="post-image"><img src="${data.relatedPost.imageUrl}" alt=""></div>` : ''}
          <div class="post-location">
            <i class="ri-map-pin-line"></i>
            <span>${data.relatedPost.locationName}</span>
          </div>
        </div>
      ` : ''}
      <div class="discovery-actions">
        <button class="add-to-plan-btn" onclick="addToPlan(${data.relatedPost?.id})">
          加入行程
        </button>
        <button class="view-on-map-btn" onclick="viewOnMap(${data.relatedPost?.lat}, ${data.relatedPost?.lng})">
          在地图上查看
        </button>
      </div>
    </div>
  `;
}
```

### 5.3 加入行程

```javascript
async function addToPlan(postId) {
  const userId = localStorage.getItem('current_user_id');
  const planId = localStorage.getItem('current_plan_id');
  
  if (!planId) {
    showToast('请先开始旅行');
    return;
  }
  
  try {
    showLoading('添加中...');
    
    const response = await fetch('/api/travel/plan/add-spot', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ userId, planId, postId })
    });
    
    const result = await response.json();
    
    if (result.success) {
      showToast('已加入行程！');
      closeMail();
    } else {
      showToast(result.error || '添加失败');
    }
  } catch (error) {
    showToast('添加失败');
  } finally {
    hideLoading();
  }
}
```

## 七、数据库

### 7.1 mails 表

```sql
CREATE TABLE mails (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    sender_type TEXT NOT NULL DEFAULT 'system',
    sender_id INTEGER,
    mail_type TEXT NOT NULL,
    title TEXT NOT NULL,
    content TEXT NOT NULL,
    image_url TEXT,
    extra_data TEXT,
    is_read INTEGER DEFAULT 0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_mails_user_id ON mails(user_id);
CREATE INDEX idx_mails_type ON mails(mail_type);
CREATE INDEX idx_mails_created ON mails(created_at);
```

## 八、AI分身探索服务（玩法1）

### 8.1 触发时机

1. 用户开始旅行计划时
2. 用户手动"空投分身"到某个城市
3. 每天旅行结束时（探索明日目的地）

### 8.2 探索逻辑

```javascript
async function agentExplore(userId, city) {
  // 1. 获取用户记忆
  const memories = await getUserMemories(userId);
  const preferences = extractPreferences(memories);
  
  // 2. 获取城市历史帖子
  const posts = await getCityPosts(city);
  
  // 3. AI 匹配分析
  const matches = [];
  for (const post of posts) {
    const score = await analyzeMatch(preferences, post);
    if (score > 0.7) {
      matches.push({ post, score, reason: await generateReason(preferences, post) });
    }
  }
  
  // 4. 生成邮件
  for (const match of matches.slice(0, 3)) {
    await createDiscoveryMail(userId, match);
  }
}

async function createDiscoveryMail(userId, match) {
  const { post, reason } = match;
  
  const timeAgo = calculateTimeAgo(post.createdAt);
  
  const content = `你的分身在${post.locationName}闲逛时，被一条<strong>${timeAgo}</strong>的帖子【${post.title || truncate(post.content, 10)}】主动搭讪了，因为<strong>${reason}</strong>。`;
  
  await db.prepare(`
    INSERT INTO mails (user_id, mail_type, title, content, extra_data)
    VALUES (?, 'discovery', '分身发现了新地点', ?, ?)
  `).run(userId, content, JSON.stringify({
    relatedPost: {
      id: post.id,
      username: post.username,
      content: post.content,
      imageUrl: post.imageUrl,
      locationName: post.locationName,
      lat: post.lat,
      lng: post.lng,
      timeAgo
    }
  }));
}
```
