# 贴文发布设计

**版本**: MVP v1.0  
**优先级**: P0 - 核心功能  
**依赖**: 地图首页、登录模块

---

## 一、功能概述

用户在旅行过程中发布贴文（文字/图片），贴文显示在地图上，并触发城市聊天室匹配。

---

## 二、用户流程

```
用户在旅行中点击"+"按钮
    ↓
选择发布类型
├── 纯文字 → 输入内容 → 选择心情 → 发布
└── 图片 → 拍照/选择图片 → 添加描述 → 选择心情 → 发布
    ↓
获取当前地理位置（经纬度 + 地区名称）
    ↓
保存贴文到数据库
    ↓
地图上显示表情标记
    ↓
检测周围 1km 内的历史贴文
    ↓
├── 有 → 创建群聊，跳转到 Spark 页面
└── 无 → 提示"你是第一个在这里发帖的人"
```

## 四、前端逻辑

### 4.1 获取当前位置

```javascript
async function getCurrentLocation() {
  return new Promise((resolve, reject) => {
    if (!navigator.geolocation) {
      reject(new Error('浏览器不支持定位'));
      return;
    }
    
    navigator.geolocation.getCurrentPosition(
      async (position) => {
        const lat = position.coords.latitude;
        const lng = position.coords.longitude;
        
        // 使用高德逆地理编码获取地址
        const address = await reverseGeocode(lat, lng);
        
        resolve({
          lat,
          lng,
          city: address.city,
          district: address.district,
          locationName: address.formattedAddress
        });
      },
      (error) => {
        reject(error);
      },
      { enableHighAccuracy: true }
    );
  });
}

async function reverseGeocode(lat, lng) {
  const response = await fetch(
    `https://restapi.amap.com/v3/geocode/regeo?key=${AMAP_KEY}&location=${lng},${lat}`
  );
  const data = await response.json();
  
  if (data.status === '1' && data.regeocode) {
    const addr = data.regeocode.addressComponent;
    return {
      city: addr.city || addr.province,
      district: addr.district,
      formattedAddress: data.regeocode.formatted_address
    };
  }
  
  return { city: '未知', district: '', formattedAddress: '未知位置' };
}
```

### 4.2 发布文字贴文

```javascript
async function publishTextPost() {
  const content = document.getElementById('postContent').value.trim();
  const mood = document.querySelector('.mood-btn.selected')?.dataset.mood;
  
  if (!content) {
    showToast('请输入内容');
    return;
  }
  
  try {
    showLoading('发布中...');
    
    const location = await getCurrentLocation();
    const userId = localStorage.getItem('current_user_id');
    const planId = localStorage.getItem('current_plan_id');
    
    const response = await fetch('/api/posts', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        userId,
        planId,
        content,
        mood,
        city: location.city,
        district: location.district,
        locationName: location.locationName,
        lat: location.lat,
        lng: location.lng
      })
    });
    
    const result = await response.json();
    
    if (result.success) {
      showToast('发布成功！');
      
      // 添加地图标记
      addPostMarker(result.data);
      
      // 检查是否触发群聊
      await checkAndCreateChatroom(result.data);
      
      hideTextPost();
    }
  } catch (error) {
    showToast('发布失败');
  } finally {
    hideLoading();
  }
}
```

### 4.3 发布图片贴文

```javascript
async function publishImagePost() {
  const imageInput = document.getElementById('imageInput');
  const caption = document.getElementById('imageCaption').value.trim();
  const mood = document.querySelector('.mood-btn.selected')?.dataset.mood;
  
  if (!imageInput.files[0]) {
    showToast('请选择图片');
    return;
  }
  
  try {
    showLoading('上传中...');
    
    // 1. 上传图片
    const formData = new FormData();
    formData.append('image', imageInput.files[0]);
    
    const uploadResponse = await fetch('/api/upload/image', {
      method: 'POST',
      body: formData
    });
    
    const uploadResult = await uploadResponse.json();
    
    if (!uploadResult.success) {
      throw new Error('图片上传失败');
    }
    
    // 2. 获取位置
    const location = await getCurrentLocation();
    const userId = localStorage.getItem('current_user_id');
    const planId = localStorage.getItem('current_plan_id');
    
    // 3. 发布贴文
    const response = await fetch('/api/posts', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        userId,
        planId,
        content: caption,
        imageUrl: uploadResult.data.url,
        mood,
        city: location.city,
        district: location.district,
        locationName: location.locationName,
        lat: location.lat,
        lng: location.lng
      })
    });
    
    const result = await response.json();
    
    if (result.success) {
      showToast('发布成功！');
      addPostMarker(result.data);
      await checkAndCreateChatroom(result.data);
      hideImagePost();
    }
  } catch (error) {
    showToast('发布失败');
  } finally {
    hideLoading();
  }
}
```

### 4.4 地图标记

```javascript
function addPostMarker(post) {
  const moodEmoji = {
    happy: '😊',
    excited: '🤩',
    peaceful: '😌',
    tired: '😴',
    surprised: '😮'
  };
  
  const marker = new AMap.Marker({
    position: [post.lng, post.lat],
    content: `<div class="post-marker">${moodEmoji[post.mood] || '📍'}</div>`,
    offset: new AMap.Pixel(-15, -15)
  });
  
  marker.on('click', () => {
    showPostDetail(post);
  });
  
  map.add(marker);
}
```

### 4.5 检查并创建群聊

```javascript
async function checkAndCreateChatroom(post) {
  try {
    const response = await fetch('/api/chatrooms/create-by-location', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        userId: post.userId,
        postId: post.id,
        city: post.city,
        district: post.district,
        lat: post.lat,
        lng: post.lng,
        radius: 1000
      })
    });
    
    const result = await response.json();
    
    if (result.success && result.data.matchedUsers.length > 0) {
      showToast(`发现 ${result.data.matchedUsers.length} 位附近的旅行者！`);
      
      // 跳转到 Spark 页面
      setTimeout(() => {
        window.location.href = `/pages/mobile/spark.html?chatroom=${result.data.chatroomId}`;
      }, 1500);
    } else {
      showToast('你是第一个在这里发帖的人！');
    }
  } catch (error) {
    console.error('创建群聊失败:', error);
  }
}
```

## 六、数据库

### 6.1 posts 表

```sql
CREATE TABLE posts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    plan_id INTEGER,
    content TEXT,
    image_url TEXT,
    mood TEXT,
    city TEXT NOT NULL,
    district TEXT,
    location_name TEXT,
    lat REAL NOT NULL,
    lng REAL NOT NULL,
    is_public INTEGER DEFAULT 1,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (plan_id) REFERENCES travel_plans(id)
);

CREATE INDEX idx_posts_city ON posts(city);
CREATE INDEX idx_posts_location ON posts(lat, lng);
CREATE INDEX idx_posts_user_id ON posts(user_id);
```

## 七、AI 提示词设计

### 7.1 记忆分类提示词

**来源**: `deepseek.js` - `classifyMemory()`

**用途**: 用户添加本地记忆时，AI 自动分类

**System Prompt**:
```
你是记忆分类助手。只输出JSON数组，不要其他内容。
```

**User Prompt 模板**:
```
请分析以下用户输入的记忆内容，将其拆分成多个独立的分类。

**重要规则：**
1. 必须将内容拆分成多个独立的分类对象，每个对象只包含一个主题
2. 不要把所有内容合并在一起
3. 每个分类的 content 字段只包含该分类相关的具体内容
4. 不要使用 Markdown 格式，直接用纯文本

记忆内容：
${content}

常见分类：个人信息、学业经历、工作经验、技术能力、参赛经历、兴趣爱好、性格特点、职业目标

输出格式（JSON 数组）：
[
  {"category":"个人信息","title":"基本信息","content":"出生年份：2003年，家乡：南京"},
  {"category":"学业经历","title":"留学计划","content":"2026年3月24日前往日本福井大学攻读研究生"},
  {"category":"技术能力","title":"语言能力","content":"日语N2，托福85分"}
]

请直接输出JSON数组：
```

**调用参数**:
```javascript
{
  temperature: 0.3,
  max_tokens: 2000
}
```
